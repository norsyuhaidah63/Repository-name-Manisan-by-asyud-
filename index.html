function renderReports() { const main = document.getElementById('mainContent'); const products = cached.products || []; const sales = cached.sales || []; const expenses = cached.expenses || []; const currentMonth = today().slice(0, 7); const dates = sales.map(r => r.sale_date).filter(Boolean); const expenseDates = expenses.map(r => r.expense_date).filter(Boolean); const allDates = [...dates, ...expenseDates].sort(); const defaultStart = allDates.length ? allDates[0] : today(); main.innerHTML = `

📊 Laporan dan Keuntungan

Semak prestasi perniagaan Manisan by Asyud.

Tempoh bulan semasa: ${esc(currentMonth)}

`; function calculate() { const start = document.getElementById('reportStart').value; const end = document.getElementById('reportEnd').value; const result = document.getElementById('reportResults'); if (!start || !end || start > end) { result.innerHTML = '
Sila pilih julat tarikh yang sah.
'; return; } const periodSales = sales.filter(r => r.sale_date && r.sale_date >= start && r.sale_date <= end ); const periodExpenses = expenses.filter(r => r.expense_date && r.expense_date >= start && r.expense_date <= end ); const totalSales = periodSales.reduce( (sum, r) => sum + Number(r.amount || 0), 0 ); const totalExpenses = periodExpenses.reduce( (sum, r) => sum + Number(r.amount || 0), 0 ); let knownCost = 0; let unmatchedQuantity = 0; periodSales.forEach(sale => { const product = products.find(p => String(p.name || '').trim().toLowerCase() === String(sale.product_name || '').trim().toLowerCase() ); if (product && product.cost_price !== null && product.cost_price !== undefined && product.cost_price !== '') { knownCost += Number(product.cost_price || 0) * Number(sale.quantity || 0); } else { unmatchedQuantity += Number(sale.quantity || 0); } }); const estimatedProfit = totalSales - knownCost - totalExpenses; const bestSellers = {}; periodSales.forEach(sale => { const name = sale.product_name || 'Produk tidak dinamakan'; bestSellers[name] = (bestSellers[name] || 0) + Number(sale.quantity || 0); }); const ranking = Object.entries(bestSellers) .sort((a, b) => b[1] - a[1]); result.innerHTML = `
Jumlah jualan
${money(totalSales)}
Kos produk terjual (anggaran)
${money(knownCost)}
Perbelanjaan
${money(totalExpenses)}
Anggaran untung bersih
${money(estimatedProfit)}

Ringkasan laporan

Tempoh: ${esc(start)} hingga ${esc(end)}

Bilangan rekod jualan: ${periodSales.length}

Bilangan rekod perbelanjaan: ${periodExpenses.length}

Kuantiti jualan tanpa padanan kos produk: ${unmatchedQuantity}

Untung ini ialah anggaran: jualan − kos produk terjual − perbelanjaan. Kos produk menggunakan modal semasa dalam menu Produk & Untung. Jika harga modal pernah berubah, laporan lama mungkin tidak tepat. Pastikan perbelanjaan yang direkod tidak mengandungi kos produk yang sudah dikira sekali lagi dalam modal.

🏆 Produk paling laris

${ranking.length ? `
${ranking.map(([name, qty]) => ` `).join('')}
ProdukKuantiti terjual
${esc(name)} ${esc(qty)}
` : '

Tiada rekod jualan bagi tempoh ini.

'}
`; } document.getElementById('applyReport') .addEventListener('click', calculate); calculate(); }
